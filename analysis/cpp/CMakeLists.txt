cmake_minimum_required(VERSION 3.18)
include(FetchContent)
find_package(OpenSSL 3.0 REQUIRED)
find_package(Threads REQUIRED)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install ON CACHE BOOL "" FORCE)
FetchContent_Declare(ai_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
FetchContent_Declare(ai_http GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG adf58bf474fac638160592d6c3f67da4ebc7df20 SOURCE_SUBDIR header-only)
FetchContent_Declare(ai_pdf GIT_REPOSITORY https://github.com/libharu/libharu.git GIT_TAG 0c598becaadaef8e3d12b883f9fc2864a118c12d)
FetchContent_MakeAvailable(ai_json)
# HTTP is header-only; do not build its examples or optional compression backends.
FetchContent_MakeAvailable(ai_http)
# Keep PDF dependencies isolated from the header-only core.
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(BUILD_SHARED_LIBS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_PNG ON)
set(CMAKE_DISABLE_FIND_PACKAGE_ZLIB ON)
set(LIBHPDF_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(ai_pdf)
add_library(ComtradeAI STATIC src/ai.cpp src/evidence.cpp src/pdf.cpp)
add_library(comtrade::ComtradeAI ALIAS ComtradeAI)
target_include_directories(ComtradeAI PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE ${ai_http_SOURCE_DIR} ${ai_pdf_SOURCE_DIR}/include ${ai_pdf_BINARY_DIR}/include)
target_compile_definitions(ComtradeAI PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
target_link_libraries(ComtradeAI PUBLIC comtrade::ComtradeCore nlohmann_json::nlohmann_json
    PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads hpdf)
if(WIN32)
    target_link_libraries(ComtradeAI PRIVATE ws2_32 crypt32)
endif()
if(MSVC)
    target_compile_options(ComtradeAI PRIVATE /utf-8)
endif()
add_executable(comtrade-ai src/main.cpp)
target_link_libraries(comtrade-ai PRIVATE ComtradeAI)
install(TARGETS ComtradeAI comtrade-ai hpdf EXPORT ComtradeAITargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT ComtradeAITargets NAMESPACE comtrade:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ComtradeCore)
install(FILES ../../config/ai-config.example.json ../../docs/ai-analysis.md
    DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/ComtradeCore)
install(FILES ${ai_http_SOURCE_DIR}/LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/ComtradeCore/licenses RENAME cpp-httplib.txt)
install(FILES ${ai_json_SOURCE_DIR}/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/ComtradeCore/licenses RENAME nlohmann-json.txt)
install(FILES ${ai_pdf_SOURCE_DIR}/LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/ComtradeCore/licenses RENAME libharu.txt)
if(COMTRADE_BUILD_TESTS)
    add_executable(comtrade_ai_tests tests/test_ai.cpp)
    target_include_directories(comtrade_ai_tests PRIVATE ${ai_http_SOURCE_DIR})
    target_compile_definitions(comtrade_ai_tests PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
    target_link_libraries(comtrade_ai_tests PRIVATE ComtradeAI GTest::gtest_main OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
    if(WIN32)
        target_link_libraries(comtrade_ai_tests PRIVATE ws2_32 crypt32)
    endif()
    if(MSVC)
        target_compile_options(comtrade_ai_tests PRIVATE /utf-8)
    endif()
    add_test(NAME ai-unit-integration COMMAND comtrade_ai_tests)
    set_tests_properties(ai-unit-integration PROPERTIES TIMEOUT 90 LABELS "ai;integration")
endif()
