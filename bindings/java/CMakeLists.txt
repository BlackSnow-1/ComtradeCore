# Do not request a FindSWIG component here: CMake 3.14 can locate and version
# check SWIG, while language components were added to FindSWIG later.
find_package(SWIG 4.0 REQUIRED)
find_package(Java REQUIRED COMPONENTS Development)
find_package(JNI REQUIRED)

include(UseSWIG)
include(UseJava)

set(COMTRADE_JAVA_PACKAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated/comtrade")
set(CMAKE_SWIG_OUTDIR "${COMTRADE_JAVA_PACKAGE_DIR}")
set(SWIG_OUTFILE_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated/cxx")

set_source_files_properties(comtrade.i PROPERTIES
        CPLUSPLUS ON
        USE_TARGET_INCLUDE_DIRECTORIES TRUE
        COMPILE_OPTIONS "-package;comtrade"
)

swig_add_library(ComtradeCoreJava
        TYPE SHARED
        LANGUAGE java
        SOURCES comtrade.i
)

target_include_directories(ComtradeCoreJava PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}"
        ${JNI_INCLUDE_DIRS}
)
target_link_libraries(ComtradeCoreJava PRIVATE
        comtrade::ComtradeCore
        ${JNI_LIBRARIES}
)
set_target_properties(ComtradeCoreJava PROPERTIES
        OUTPUT_NAME ComtradeCoreJava
        POSITION_INDEPENDENT_CODE ON
)

# The Java proxy files are generated by SWIG during the build rather than
# during CMake configuration. Compile them in an explicit post-SWIG command;
# add_jar() filters out generated .java files that do not exist at configure
# time on some CMake versions.
set(COMTRADE_SWIG_JAVA_SOURCES
        "${COMTRADE_JAVA_PACKAGE_DIR}/ComtradeCoreNative.java"
        "${COMTRADE_JAVA_PACKAGE_DIR}/ComtradeCoreNativeJNI.java"
        "${COMTRADE_JAVA_PACKAGE_DIR}/ComtradeNativeRecord.java"
        "${COMTRADE_JAVA_PACKAGE_DIR}/ComtradeNativeRowCallback.java"
        "${COMTRADE_JAVA_PACKAGE_DIR}/ComtradeNativeStreamReader.java"
        "${COMTRADE_JAVA_PACKAGE_DIR}/ComtradeNativeStreamWriter.java"
        "${COMTRADE_JAVA_PACKAGE_DIR}/DoubleVector.java"
        "${COMTRADE_JAVA_PACKAGE_DIR}/IntVector.java"
)
set(COMTRADE_PUBLIC_JAVA_SOURCES
        "${CMAKE_CURRENT_SOURCE_DIR}/src/main/java/comtrade/ComtradeRecord.java"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/main/java/comtrade/ComtradeStreamReader.java"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/main/java/comtrade/ComtradeStreamWriter.java"
)
set(COMTRADE_JAVA_CLASSES_DIR "${CMAKE_CURRENT_BINARY_DIR}/java-classes")
set(COMTRADE_JAVA_JAR "${CMAKE_CURRENT_BINARY_DIR}/comtrade-core-java.jar")

add_custom_command(
        OUTPUT "${COMTRADE_JAVA_JAR}"
        COMMAND "${CMAKE_COMMAND}" -E rm -rf "${COMTRADE_JAVA_CLASSES_DIR}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${COMTRADE_JAVA_CLASSES_DIR}"
        COMMAND "${Java_JAVAC_EXECUTABLE}"
                -encoding UTF-8
                -d "${COMTRADE_JAVA_CLASSES_DIR}"
                ${COMTRADE_SWIG_JAVA_SOURCES}
                ${COMTRADE_PUBLIC_JAVA_SOURCES}
        COMMAND "${Java_JAR_EXECUTABLE}"
                cf "${COMTRADE_JAVA_JAR}"
                -C "${COMTRADE_JAVA_CLASSES_DIR}" .
        DEPENDS
        ComtradeCoreJava
        ${COMTRADE_PUBLIC_JAVA_SOURCES}
        COMMENT "Compiling ComtradeCore Java proxies and creating comtrade-core-java.jar"
        VERBATIM
)
add_custom_target(ComtradeCoreJavaJar ALL DEPENDS "${COMTRADE_JAVA_JAR}")

set(COMTRADE_JAVA_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/comtrade/java")
install(TARGETS ComtradeCoreJava
        RUNTIME DESTINATION "${COMTRADE_JAVA_INSTALL_DIR}"
        LIBRARY DESTINATION "${COMTRADE_JAVA_INSTALL_DIR}"
        ARCHIVE DESTINATION "${COMTRADE_JAVA_INSTALL_DIR}"
)
install(FILES "${COMTRADE_JAVA_JAR}"
        DESTINATION "${COMTRADE_JAVA_INSTALL_DIR}"
)
