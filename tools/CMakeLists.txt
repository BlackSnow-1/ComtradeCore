add_executable(comtrade-check comtrade_check.cpp)
target_link_libraries(comtrade-check PRIVATE comtrade::ComtradeCore)
if(MSVC)
    set_property(TARGET comtrade-check PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
    target_link_options(comtrade-check PRIVATE -municode -static)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(comtrade-check PRIVATE -static)
endif()
install(TARGETS comtrade-check RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
